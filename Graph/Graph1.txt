import java.util.*;
public class Main{
    //Edge class
    static class edge{
        int src;
        int dest;
        int wt;
        
        public edge(int s,int d,int w){
            this.src=s;
            this.dest=d;
            this.wt=w;
        } }
   
  //create graph function      
static void createGraph(ArrayList<edge>graph[]){
    for(int i=0;i<graph.length;i++){
       graph[i]=new ArrayList<>(); 
    }
    
    graph[0].add(new edge(0,1,1));
    graph[0].add(new edge(0,2,1));
    
    graph[1].add(new edge(2,0,1));
    graph[1].add(new edge(1,3,1));
    
    graph[2].add(new edge(2,0,1));
    graph[2].add(new edge(2,4,1));
    
    graph[3].add(new edge(3,1,1));
    graph[3].add(new edge(3,4,1));
    graph[3].add(new edge(3,1,1));
    
    graph[4].add(new edge(4,2,1));
    graph[4].add(new edge(4,3,1));
    graph[4].add(new edge(4,5,1)); 
    
    graph[5].add(new edge(5,3,1));
    graph[5].add(new edge(5,4,1));
    graph[5].add(new edge(5,6,1));
    graph[5].add(new edge(6,5,1));
}  

//1.BFS-Triversal o(V+E)
public static void Bfs(ArrayList<edge>graph[]){ 
    Queue<Integer>q=new LinkedList<>();
    boolean vist[]=new boolean[graph.length];
    q.add(0);
    
    while(!q.isEmpty()){
        int current=q.remove();
        
        if(!vist[current]){
            System.out.print(current+" ");
            vist[current]=true;
            for(int i=0;i<graph[current].size();i++){
                edge e=graph[current].get(i);
                q.add(e.dest);
            }}} }
  
  //2.DFS-Triversal o(V+E)          
public static void Dfs(ArrayList<edge>graph[],int current,boolean vist[]){
    
    //vist
    System.out.print(current+" ");
    vist[current]=true;
    
    for(int i=0;i<graph[current].size();i++){
        edge e=graph[current].get(i);
        if(!vist[e.dest]){
            Dfs(graph,e.dest,vist);
        }}}  
        
//3.Has path(for given sec and dest tell if path exit from src to dest)
 //Tc: o(V+E)
 public static boolean HasPath(ArrayList<edge>graph[],int src,int dest,boolean vist[]){
     
     if(src==dest){
        return true; 
     }
     
     vist[src]=true;
     for(int i=0;i<graph[src].size();i++){
         edge e=graph[src].get(i);
         if(!vist[e.dest] && HasPath(graph,e.dest,dest,vist)){
            return true; 
         }}
         return false;
 }
 
    public static void main(String args[]){
        int v=7;
        @SuppressWarnings("unchecked")
      ArrayList<edge>graph[]=new ArrayList[v];
      createGraph(graph);
      
       Bfs(graph);

      Dfs(graph,0,new boolean[v]);
      
      System.out.println(HasPath(graph,0,5,new boolean[v]) );
      
    }}
