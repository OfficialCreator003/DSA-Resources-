public class main{
 public static void printmyarray(int array[]){
     for(int i=0;i<array.length;i++){
         System.out.print(array[i]+" ");
         
     } }
     
     public static void mergesort(int array[],int startindex,int endindex){
         
     if(startindex >= endindex){
        return; 
     } 
     
          //OR:-  (startindex+endindex)/2; 
         int mid=startindex + (endindex - startindex) / 2;  
         
         mergesort(array,startindex,mid);
         mergesort(array,mid+1,endindex);
         merge(array,startindex,mid,endindex);
         }
         
  public static void merge(int array[],int startindex,int mid,int endindex){
      int temp[]=new int[endindex-startindex+1];
      int i=startindex;
      int j=mid+1;
      int k=0;
      
      while(i<=mid && j<=endindex){
          if(array[i]<array[j]){
              temp[k]=array[i];
              i++;
          } else {
              temp[k]=array[j];
              j++;
          }
          k++;   }
          //left part
          while(i<=mid){
              temp[k++]=array[i++];
          }
          //right part
          while(j<=endindex){
              temp[k++]=array[j++];
          }
         
         //copy temp to oruganal
         for (k = 0, i = startindex; k < temp.length; k++, i++) {
            array[i] = temp[k];
        }
          
  }         
public static void main(String args[]){
  int array[]={10,23,5,9,88,9};
 
 mergesort(array,0,array.length-1);
printmyarray(array);
 
}

2.//quick sort

public class main {
    public static void printmyarray(int array[]) {
        for (int val : array) {
            System.out.print(val + " ");
        }
        System.out.println();
    }

    public static void quicksort(int array[], int startindex, int endindex) {
        // 1. Base Case
        if (startindex >= endindex) {
            return;
        }

        // 2. Partition and get Pivot Index
        int pivotindex = partition(array, startindex, endindex);

        // 3. Recursive calls
        quicksort(array, startindex, pivotindex - 1); // Left part
        quicksort(array, pivotindex + 1, endindex);  // Right part
    }

    public static int partition(int array[], int startindex, int endindex) {
        int pivot = array[endindex];
        int i = startindex - 1; // Makes space for elements smaller than pivot

        for (int j = startindex; j < endindex; j++) {
            if (array[j] <= pivot) {
                i++;
                // Swap array[i] and array[j]
                int temp = array[j];
                array[j] = array[i];
                array[i] = temp;
            }
        }

        // Final swap to put pivot in correct position (at i + 1)
        i++;
        int temp = pivot;
        array[endindex] = array[i]; 
        array[i] = temp;
        
        return i; // Return the pivot index
    }

    public static void main(String args[]) {
        int array[] = {10, 20, 5, 9};
        quicksort(array, 0, array.length - 1);
        printmyarray(array);
    }
}

public class main{
    public static int sortedrotededarray(int array[],int target,int startindex,int endindex){
        
        //Base case
        if(startindex>endindex){
            return -1;
        }
       
       //kam 
      int mid=startindex+(endindex-startindex)/2;
        
        //case found
        if(array[mid]==target){
            return mid;  }
        
            
    //mid on line1    
    if(array[startindex]<=array[mid]){
       //case a: Left
       if(array[startindex]<=target && target <= array[mid]){
           return sortedrotededarray(array,target,startindex,mid-1);
           }else{
               //case b: Right
               return sortedrotededarray(array,target,mid+1,endindex);
           }  }
       
       
       //mid on line2
       else{
          //case c: Right 
          if(array[mid]<=target && target<=array[endindex] ){
              return sortedrotededarray(array,target,mid+1,endindex);
          } else {
              
              //case d: left
              return sortedrotededarray(array,target,startindex,mid-1);
          }
          
       }
    }
    
    public static void main(String args[]){
        int array[]={10,20,30,40,50};
        int target=30;
        
        int tarindex=sortedrotededarray(array,target,0,array.length-1);
        System.out.println(tarindex);
        
    }
}
