//compress the string
public class string{
    public static String compress(String str){
        String newStr="";
        
        for(int i=0;i<str.length();i++){
           Integer count=1;
          while(i<str.length()-1 && str.charAt(i)==str.charAt(i+1)){
              count ++;
              i++;
              } 
              
    newStr += str.charAt(i);
    if(count>1){
        newStr +=count.toString();
    }
        }
        
        return newStr;
    }
    public static void main(String args[]){
      String str="aaaabbbbbcccccddeeff";
      
      System.out.println(compress(str));
    }
}

convert latter to uppercase
public class string {
    public static String toupper(String str) {
        StringBuilder sb = new StringBuilder(""); 

        
        char ch = Character.toUpperCase(str.charAt(0));
        sb.append(ch);

        for (int i = 1; i < str.length(); i++) {
            
            if (str.charAt(i) == ' ' && i < str.length() - 1) {
                sb.append(str.charAt(i)); // Append the space
                i++; // Move to the character after the space
                sb.append(Character.toUpperCase(str.charAt(i)));
            } else {
                sb.append(str.charAt(i));
            }
        }
        
        return sb.toString(); 
    }

    public static void main(String args[]) {
        String str = "hii how are you";
        System.out.println(toupper(str)); 
    }
}



//shortest Path
public class string{
 public static float shhotestpath(String path){
    int x=0,y=0;
    
    for(int i=0;i<path.length();i++){
       char dir=path.charAt(i); 
       
       //south
       if(dir=='s'){
         y--; 
         
         //north
       }else if(dir=='n'){
          y++;
       }
       
       //west
       else if(dir=='w'){
          x--; 
       }
       
       //east
       else{
           x++;
       }
    }
    int x2=x*x;
    int y2=y*y;
    
    return (float)Math.sqrt(x2+y2);
    

}
public static void main(String args[]){
    String path="wneenesennn";
    
   System.out.print(shhotestpath(path) );
}
}

//bubble short(O(n^2))
public class shorting{
    public static void BubbleShort(int array[]){
        for(int turn=0;turn<array.length-1;turn++){
    for(int j=0;j<array.length-1-turn;j++){
        if(array[j]>array[j+1]){
            
            //swapping
            int temp=array[j];
            array[j]=array[j+1];
            array[j+1]=temp;
        }
    }        
        }
    }
    public static void printArray(int array[]){
        for(int i=0;i<array.length;i++){
            System.out.print(" "+array[i]);
        }
    }
    public static void main(String args[]){
        int  array[]={100,60,180,99,89};
         BubbleShort(array);
           printArray(array);
    }
}


//SlectionShort(O(n^2))
public class shorting{
    public static void SlectionShort(int array[]){
      for(int i=0;i<array.length;i++){
          int minPosition=i;
          for(int j=i+1;j<array.length;j++){
              if(array[minPosition] > array[j]){
                  minPosition=j;
              } }
              
              //Swapping
              int temp=array[minPosition];
              array[minPosition]=array[i];
              array[i]=temp;
          }  }
    
    public static void printArray(int array[]){
        for(int i=0;i<array.length;i++){
            System.out.print(" "+array[i]);
        }
    }

    public static void main(String args[]){
        int  array[]={100,60,180,99,89};
         SlectionShort(array);
         printArray(array);
    }
}

//Buy at the lowest point and sell at the highest point after that low point.
public class array2 {
    public static int buyAndSellStocks(int prices[]) {
        int buyPrice = Integer.MAX_VALUE; // Step 1: Track lowest buying price
        int maxProfit = 0;                // Step 2: Track highest profit
        
        for (int i = 0; i < prices.length; i++) {
            if (buyPrice < prices[i]) {   // Step 3: If price is higher, check profit
                int profit = prices[i] - buyPrice; 
                maxProfit = Math.max(maxProfit, profit); // Step 4: Update best profit
            } else {
                buyPrice = prices[i];     // Step 5: If price is lower, update buy price
            }
        }
        return maxProfit; // Step 6: Return final max profit
    }

    public static void main(String args[]) {
        int prices[] = {7, 1, 5, 3, 6, 4};
        System.out.println(buyAndSellStocks(prices));
    }
}




//RainTrapped Water(o(n))    
public class array2  {
    public static int RainTrappedWater(int height[]) {
        int n = height.length;

        // 1. Calculate left max boundary
        int leftMax[] = new int[n];
        leftMax[0] = height[0];
        for (int i = 1; i < n; i++) {
            leftMax[i] = Math.max(height[i], leftMax[i - 1]);
        }

        // 2. Calculate right max boundary
        int rightMax[] = new int[n];
        rightMax[n - 1] = height[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            rightMax[i] = Math.max(height[i], rightMax[i + 1]);
        }

        // 3. Final calculation loop
        int trappedWater = 0;
        for (int i = 0; i < n; i++) {
            // 4. waterLevel = min(leftmax, rightmax)
            int waterLevel = Math.min(leftMax[i], rightMax[i]);

            // 5. trapped water = waterLevel - height[i]
            trappedWater += waterLevel - height[i];
        }

        return trappedWater;
    }

    public static void main(String args[]) {
        
        int height[] = {4, 2, 0, 6, 3, 2, 5}; 
        System.out.println(RainTrappedWater(height));
    }
}

public class Main {
    public static void decToBin(int n) {
        int myNum = n;
        int pow = 0;
        int binNum = 0;
        
        while (n > 0) {
            int rem = n % 2;
            // Build the binary number using powers of 10
            binNum = binNum + (rem * (int)Math.pow(10, pow));
            pow++;
            n = n / 2;
        }
        
        // Corrected: Print only the binNum
        System.out.println("Binary of " + myNum + " is: " + binNum);
    }

    public static void main(String args[]) {
        decToBin(15); // Expected Output: 1111
        decToBin(7);  // Expected Output: 111
    }
}

Binary search
public class array{
    public static int myBin(int arr[],int tar){
        int start=0;
        int end=arr.length-1;
        
        while(start<=end){
            int mid=(start+end)/2;
            
            //compression 
            if(arr[mid]==tar){
                return mid;
            }
            if(arr[mid]<tar){
               start=mid+1; 
            } else{
                end=mid-1;
            }
        }
        return -1;
    }
    public static void main(String args[]){
        int arr[]={10,40,60,80,90,900};
        int tar=60;
        
        System.out.println("index: " + myBin(arr,tar));
    }
    
}

public class array{
    public static  void myfunction(int numbers[] ){
        int first=0;
        int last=numbers.length-1;
        while(first<last){
            //swaping
            int temp=numbers[last];
           numbers[last]=numbers[first];
           numbers[first]=temp;
           
           first++;
           last--;
        }
        
    }
    public static void main(String args[]){
        int numbers[]={10,80,70,81};
        
        myfunction(numbers);
        for(int i=0;i<numbers.length;i++){
            System.out.println(numbers[i]);
        }
        

    }
}
