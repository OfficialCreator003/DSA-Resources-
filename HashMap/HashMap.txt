//2.Custom HashMap Implementation in Java
public class CustomHashMap<K, V> {
    
    // 1. कस्टम नोड क्लास (Linked List के लिए)
    private class Node {
        K key;
        V value;
        Node next;
        
        public Node(K key, V value) {
            this.key = key;
            this.value = value;
            this.next = null;
        }
    }

    private Node[] buckets;
    private int size; // कुल कितने elements हैं
    private int capacity = 4; // Array का शुरुआती साइज (उदाहरण के लिए छोटा रखा है)
    private final double LOAD_FACTOR_THRESHOLD = 0.75;

    // 2. Constructor
    @SuppressWarnings("unchecked")
    public CustomHashMap() {
        buckets = new CustomHashMap.Node[capacity];
        size = 0;
    }

    // 3. Hash Function (Key को Index में बदलना)
    private int getBucketIndex(K key) {
        int hashCode = key.hashCode();
        return Math.abs(hashCode) % capacity; // 0 से capacity-1 तक का इंडेक्स
    }

    // 4. PUT Method (डेटा डालना)
    public void put(K key, V value) {
        int bi = getBucketIndex(key);
        Node head = buckets[bi];
        Node current = head;

        // Case 1: अगर Key पहले से मौजूद है, तो बस Value अपडेट करें
        while (current != null) {
            if (current.key.equals(key)) {
                current.value = value;
                return;
            }
            current = current.next;
        }

        // Case 2: Key नहीं है, तो नया नोड बनाएं और Linked List की शुरुआत में जोड़ दें
        Node newNode = new Node(key, value);
        newNode.next = head;
        buckets[bi] = newNode;
        size++;

        // Rehashing: अगर डेटा बहुत ज्यादा हो जाए, तो Array का साइज दोगुना कर दें
        double currentLoadFactor = (double) size / capacity;
        if (currentLoadFactor > LOAD_FACTOR_THRESHOLD) {
            rehash();
        }
    }

    // 5. GET Method (डेटा निकालना)
    public V get(K key) {
        int bi = getBucketIndex(key);
        Node current = buckets[bi];

        // Linked list में Key को ढूंढें
        while (current != null) {
            if (current.key.equals(key)) {
                return current.value;
            }
            current = current.next;
        }
        return null; // Key नहीं मिली
    }

    // 6. REMOVE Method (डेटा हटाना)
    public V remove(K key) {
        int bi = getBucketIndex(key);
        Node current = buckets[bi];
        Node prev = null;

        while (current != null) {
            if (current.key.equals(key)) {
                if (prev == null) {
                    // अगर पहला ही नोड डिलीट करना है
                    buckets[bi] = current.next;
                } else {
                    // बीच का या आखिरी नोड डिलीट करना है
                    prev.next = current.next;
                }
                size--;
                return current.value;
            }
            prev = current;
            current = current.next;
        }
        return null;
    }

    // 7. Rehashing Method (जब Bucket भर जाएं)
    @SuppressWarnings("unchecked")
    private void rehash() {
        Node[] oldBuckets = buckets;
        capacity *= 2; // साइज दोगुना करें
        buckets = new CustomHashMap.Node[capacity];
        size = 0; // put मेथड इसे वापस बढ़ाएगा

        // पुराने एरे के हर बकेट की हर लिंक्ड लिस्ट को नए एरे में शिफ्ट करें
        for (Node head : oldBuckets) {
            Node current = head;
            while (current != null) {
                put(current.key, current.value);
                current = current.next;
            }
        }
    }

    public int size() {
        return size;
    }
    
    // Main Method Test करने के लिए
    public static void main(String[] args) {
        CustomHashMap<String, Integer> map = new CustomHashMap<>();
        
        map.put("India", 140);
        map.put("China", 141);
        map.put("USA", 33);
        
        System.out.println("India's Population: " + map.get("India"));
        System.out.println("Size before remove: " + map.size());
        
        map.remove("China");
        System.out.println("Size after remove: " + map.size());
        System.out.println("China exists? : " + map.get("China"));
    }
}

//4.Valid Anagram  o(n)
import java.util.*;
public class Main{
    public static boolean ValidAnagram(String s,String t){
         HashMap<Character,Integer>map=new HashMap<>();
         for(int i=0;i<s.length();i++){
        char ch=s.charAt(i);
        map.put(ch,map.getOrDefault(ch,0)+1);
         }
         
         for(int i=0;i<t.length();i++){
             char ch=t.charAt(i);
             if(map.get(ch)!=null){
                 if(map.get(ch)==1){
                     map.remove(ch);
                 }else{
                     map.put(ch,map.get(ch)-1);
                 }}else{
                     return false;
                 }}
        return map.isEmpty(); 
    }

    public static void main(String args[]){
       String s="race";
       String t="care";
        
        System.out.println(ValidAnagram(s,t));
    } }

//Union and InterSection
//TC:- o(n+m)  where; n= arr1_size , m=arr2_size .
import java.util.*;
public class Main{
    public static void main(String args[]){
        int arr1[]={7,3,9};
        int arr2[]={6,3,9,2,9,4};
      HashSet<Integer>Set=new HashSet<>();
      
      //1.Union
      for(int i=0;i<arr1.length;i++){
         Set.add(arr1[i]); }
         
      for(int i=0;i<arr2.length;i++){
          Set.add(arr2[i]); }
          
      System.out.println("Union: "+ Set.size());
      
      //2.InterSection
      Set.clear();
      
      for(int i=0;i<arr1.length;i++){
          Set.add(arr1[i]); }
          
      int count=0;
      for(int i=0;i<arr2.length;i++){
          if(Set.contains(arr2[i])){
              count++;
              Set.remove(arr2[i]);
          }}
    System.out.println("InterSection: "+count);
    
    }}







//Find Itinerary Ticket o(N)
import java.util.*;
public class Main{
    public static String getStart(HashMap<String,String>ticket){
        HashMap<String,String>revMap=new HashMap<>();
        
        for(String key: ticket.keySet()){
            revMap.put(ticket.get(key),key);
        }
        
        for(String key: ticket.keySet()){
            if(!revMap.containsKey(key)){
                return key;  //starting point
            }}
            return null;
    }
    public static void main(String args[]){
        HashMap<String,String>ticket=new HashMap<>();
        
        ticket.put("Channeai","Bangaluru");
        ticket.put("Mumbi","Delhi");
        ticket.put("Goa","Channeai");
        ticket.put("Delhi","Goa");
        
        String start=getStart(ticket);
        System.out.println(start);
        
        for(String key:ticket.keySet()){
            System.out.print("->"+ticket.get(start));
            start=ticket.get(start);
        }}}
