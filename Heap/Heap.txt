//2. PriorityQueue for objects

import java.util.PriorityQueue;
public class Main {
    
    // Capitalized the class name to follow Java standards
    static class Student implements Comparable<Student> {
        String name;
        int rank;
        
        public Student(String name, int rank) {
            this.name = name;
            this.rank = rank;
        }

        @Override
        public int compareTo(Student s2) {
            // Sorts in ascending order based on rank
            return this.rank - s2.rank; 
        }}

    public static void main(String args[]) {
        PriorityQueue<Student> pq = new PriorityQueue<>();

        
        pq.add(new Student("A", 4));
        pq.add(new Student("B", 1)); 
        pq.add(new Student("C", 3));

        while (!pq.isEmpty()) {
            System.out.println("Student: " + pq.peek().name + " - Rank: " + pq.peek().rank);
            pq.remove();
        }}}





//2. create heap using ArrayList and add,peek and remove method
import java.util.ArrayList;
public class Main {
    static class Heap {
        ArrayList<Integer> arr = new ArrayList<>();

        //1.add method
        public void add(int data) {
            arr.add(data);

            int x = arr.size() - 1; // x is child index
            int parent = (x - 1) / 2; // parent index

            while (x > 0 && arr.get(x) < arr.get(parent)) {
                // Swap
                int temp = arr.get(x);
                arr.set(x, arr.get(parent));
                arr.set(parent, temp);
                
                // Move this INSIDE the loop so it keeps checking upwards
                x = parent;
                parent = (x - 1) / 2;
            }}
 
         //2.peek method
        public int peek() {
            return arr.get(0); 
        }
     
        //3.remove method
        public int remove() {
            int data = arr.get(0);

            // STEP 1: Swap first and last
            int temp = arr.get(0);
            arr.set(0, arr.get(arr.size() - 1));
            arr.set(arr.size() - 1, temp);

            // STEP 2: Delete last
            arr.remove(arr.size() - 1);

            // STEP 3: Heapify
            if (!isEmpty()) {
                Heapify(0); 
            }
            return data;
        }
        
        //Helper function for remove method
        private void Heapify(int i) {
            int left = 2 * i + 1;
            int right = 2 * i + 2;
            int minIndex = i;

            //Changed && to > to actually compare the values
            if (left < arr.size() && arr.get(minIndex) > arr.get(left)) {
                minIndex = left;
            }

            if (right < arr.size() && arr.get(minIndex) > arr.get(right)) {
                minIndex = right;
            }

            //Compare to 'i', not '0'
            if (minIndex != i) {
                // Swap
                int temp = arr.get(i);
                arr.set(i, arr.get(minIndex));
                arr.set(minIndex, temp);

                // Recursively fix the affected sub-tree
                Heapify(minIndex);
            }}
    
          //check empty then return
        public boolean isEmpty() {
            return arr.size() == 0;
        }}

    public static void main(String args[]) {
        Heap h = new Heap();
        h.add(3);
        h.add(4);
        h.add(1);
        h.add(5);

        while (!h.isEmpty()) {
            System.out.print(h.peek() + " ");
            h.remove();
        }}}
